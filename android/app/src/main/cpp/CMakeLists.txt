cmake_minimum_required(VERSION 3.22.1)
project(investment_radar_android LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(IR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json URL https://codeload.github.com/nlohmann/json/tar.gz/9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03 URL_HASH SHA256=0dbc5e40a01ff142e7e68c03e85247a4dcede2f592d12d3677dee3664d17975a DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(json)
add_library(investment_core STATIC ${IR_ROOT}/core/src/investment_engine.cpp ${IR_ROOT}/core/src/provider_manager.cpp ${IR_ROOT}/core/src/providers/http_providers.cpp ${IR_ROOT}/core/src/research_service.cpp)
target_include_directories(investment_core PUBLIC ${IR_ROOT}/core/include)
target_link_libraries(investment_core PUBLIC nlohmann_json::nlohmann_json)
add_library(investment_radar SHARED native_bridge.cpp)
target_link_libraries(investment_radar investment_core android log)
