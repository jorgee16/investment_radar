cmake_minimum_required(VERSION 3.22)
project(investment_radar VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(IR_BUILD_DESKTOP "Build the native WebView shell (GTK/WebKit on Linux)" ON)
if(MSVC)
  add_compile_options(/utf-8)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
include(FetchContent)
include(CTest)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
set(HTTPLIB_REQUIRE_OPENSSL ON CACHE BOOL "" FORCE)
set(HTTPLIB_USE_BROTLI_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_ZSTD_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(httplib GIT_REPOSITORY https://github.com/yhirose/cpp-httplib GIT_TAG 787a34ad7f01f20922a237d5142aae469828be72)
FetchContent_Declare(json URL https://codeload.github.com/nlohmann/json/tar.gz/9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03 URL_HASH SHA256=0dbc5e40a01ff142e7e68c03e85247a4dcede2f592d12d3677dee3664d17975a DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(httplib json)
add_library(investment_core core/src/investment_engine.cpp core/src/provider_manager.cpp core/src/providers/http_providers.cpp core/src/research_service.cpp)
target_include_directories(investment_core PUBLIC core/include)
target_link_libraries(investment_core PUBLIC nlohmann_json::nlohmann_json)
add_library(investment_platform desktop/src/platform_services.cpp)
target_include_directories(investment_platform PUBLIC desktop/src)
if(WIN32)
  target_link_libraries(investment_platform PUBLIC crypt32)
endif()
target_link_libraries(investment_platform PUBLIC investment_core httplib::httplib OpenSSL::SSL OpenSSL::Crypto SQLite::SQLite3)
add_executable(investment-radar desktop/src/main.cpp)
target_link_libraries(investment-radar PRIVATE investment_platform Threads::Threads)

# Headless native host used by the Compose desktop UI. It owns provider routing,
# secure config/cache and the C++ investment engine; Compose remains a pure UI client.
add_executable(investment-radar-core-host desktop/src/main.cpp)
target_link_libraries(investment-radar-core-host PRIVATE investment_platform Threads::Threads)
if(MSVC)
  # Ship the desktop app as a GUI-subsystem executable so Windows does not
  # allocate a console window when the user launches investment-radar.exe.
  # Keep main() as the entry point because the same binary also supports
  # --headless for CI/smoke tests.
  target_link_options(investment-radar PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:mainCRTStartup)
  target_link_options(investment-radar-core-host PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:mainCRTStartup)
endif()
if(IR_BUILD_DESKTOP)
  FetchContent_Declare(webview GIT_REPOSITORY https://github.com/webview/webview GIT_TAG 3ab4b5d722438fc8a13e6ca830c5e2372d19a01d)
  FetchContent_MakeAvailable(webview)
  target_link_libraries(investment-radar PRIVATE webview::core)
  target_compile_definitions(investment-radar PRIVATE IR_WITH_WEBVIEW)
endif()
add_custom_target(investment_assets ALL COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/shared/web $<TARGET_FILE_DIR:investment-radar>/assets)
add_dependencies(investment-radar investment_assets)
add_dependencies(investment-radar-core-host investment_assets)
# Runtime packages contain only the app, its assets, libraries and notices.
# FetchContent dependencies may also define install rules; use --component Runtime.
install(TARGETS investment-radar investment-radar-core-host RUNTIME DESTINATION . COMPONENT Runtime)
install(DIRECTORY shared/web/ DESTINATION assets COMPONENT Runtime)
install(FILES docs/RUNNING.md DESTINATION . COMPONENT Runtime RENAME README.md)
install(FILES "${httplib_SOURCE_DIR}/LICENSE" DESTINATION licenses COMPONENT Runtime RENAME cpp-httplib.txt)
install(FILES "${json_SOURCE_DIR}/LICENSE.MIT" DESTINATION licenses COMPONENT Runtime RENAME nlohmann-json.txt)
if(IR_BUILD_DESKTOP)
  install(FILES "${webview_SOURCE_DIR}/LICENSE" DESTINATION licenses COMPONENT Runtime RENAME webview.txt)
endif()
if(WIN32)
  # vcpkg copies transitive OpenSSL/SQLite DLLs next to the built executable.
  install(DIRECTORY "$<TARGET_FILE_DIR:investment-radar>/" DESTINATION . COMPONENT Runtime
    FILES_MATCHING PATTERN "*.dll" PATTERN "assets" EXCLUDE)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_COMPONENT Runtime)
  include(InstallRequiredSystemLibraries)
  if(DEFINED VCPKG_INSTALLED_DIR)
    foreach(dependency openssl sqlite3)
      install(FILES "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/${dependency}/copyright"
        DESTINATION licenses COMPONENT Runtime RENAME "${dependency}.txt")
    endforeach()
  endif()
endif()
if(BUILD_TESTING)
  add_executable(core_tests core/test.cpp)
  target_link_libraries(core_tests PRIVATE investment_core)
  add_test(NAME core COMMAND core_tests)
  add_executable(platform_tests desktop/test.cpp)
  target_link_libraries(platform_tests PRIVATE investment_platform)
  add_test(NAME platform COMMAND platform_tests)
endif()
